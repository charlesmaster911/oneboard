왜 백업? OneBoard PostgreSQL이 장애나 실수로 데이터가 사라져도, Google Sheet에 어제 자정까지의 데이터가 안전하게 보관됩니다. 읽기 전용 백업이라 시트에서 직접 편집해도 OneBoard에 반영되진 않습니다 (단방향).

단계 1 — Apps Script 설정 (시트는 이미 만들어둠)

2 기본 코드 모두 지우고 아래 코드 붙여넣기: 
// OneBoard → Google Sheet 자동 백업
// 매일 새벽 03:00 KST 자동 실행 (트리거 설정 후)

const ONEBOARD_API = 'https://oneboard-server.onrender.com/api/team/export';
const ADMIN_TOKEN = 'JEWELICE';  // OneBoard 관리자 토큰

function backup() {
  try {
    const res = UrlFetchApp.fetch(ONEBOARD_API, {
      method: 'get',
      headers: { 'X-Admin-Token': ADMIN_TOKEN },
      muteHttpExceptions: true,
    });
    if (res.getResponseCode() !== 200) {
      Logger.log('백업 실패: HTTP ' + res.getResponseCode() + ' — ' + res.getContentText());
      return;
    }
    const data = JSON.parse(res.getContentText());
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // ── tasks 시트 덮어쓰기 ──
    let tasksSheet = ss.getSheetByName('tasks');
    if (!tasksSheet) tasksSheet = ss.insertSheet('tasks');
    tasksSheet.clear();
    tasksSheet.appendRow(['date','assignee','task','status','priority','memo','created_at','updated_at']);
    if (data.tasks && data.tasks.length) {
      const rows = data.tasks.map(t => [
        t.date, t.assignee, t.task, t.status, t.priority,
        t.memo || '', t.created_at, t.updated_at
      ]);
      tasksSheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
    }

    // ── minutes 시트 덮어쓰기 ──
    let minutesSheet = ss.getSheetByName('minutes');
    if (!minutesSheet) minutesSheet = ss.insertSheet('minutes');
    minutesSheet.clear();
    minutesSheet.appendRow(['date','title','attendees','status','directives','directive_states','content','created_at']);
    if (data.minutes && data.minutes.length) {
      const rows = data.minutes.map(m => [
        m.date, m.title, m.attendees,
        m.status,
        m.directives || '',
        JSON.stringify(m.directive_states || []),
        m.content || '',
        m.created_at
      ]);
      minutesSheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
    }

    // ── 메타 정보 (A1 셀) ──
    let metaSheet = ss.getSheetByName('_meta');
    if (!metaSheet) metaSheet = ss.insertSheet('_meta');
    metaSheet.clear();
    metaSheet.appendRow(['exportedAt', 'tasks_count', 'minutes_count']);
    metaSheet.appendRow([data.exportedAt, (data.tasks||[]).length, (data.minutes||[]).length]);

    Logger.log('✅ 백업 완료: tasks ' + (data.tasks||[]).length + ', minutes ' + (data.minutes||[]).length);
  } catch (err) {
    Logger.log('⚠ 백업 오류: ' + err.message);
  }
}
4 상단 저장 (💾) 후 함수 선택 칸에서 backup 선택 → 실행 한 번 (권한 승인 필요)
5 첫 실행에서 권한 승인 화면 뜨면 → "고급" → "신뢰할 수 있음" → 진행

문제 해결

증상원인 / 조치
HTTP 401 / 403ADMIN_TOKEN 오타. 정확히 JEWELICE 인지 확인
HTTP 500Render 백엔드 일시 장애. 다음 실행 대기 또는 수동 재실행
tasks/minutes 비어있음OneBoard에 아직 DB 저장된 데이터가 없음. PRESET ghost(점선) 항목은 export 안 됨 — "영구 저장" 한 번 클릭
권한 오류Apps Script 첫 실행 시 권한 승인 필수. 다시 실행 → "고급" → 진행
